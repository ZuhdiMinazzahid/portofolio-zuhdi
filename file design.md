  function filterProjects(category) {
    const cards = document.querySelectorAll('.project-card');

    // Update active button styles
    buttons.forEach(btn => {
      if (btn.getAttribute('data-filter-btn') === category) {
        btn.classList.add('bg-primary-container', 'text-on-secondary-fixed', 'font-bold', 'shadow');
        btn.classList.remove('text-secondary');
      } else {
        btn.classList.remove('bg-primary-container', 'text-on-secondary-fixed', 'font-bold', 'shadow');
        btn.classList.add('text-secondary');
      }
    });

    // Filter project items
    cards.forEach(card => {
      const cardCategory = card.getAttribute('data-category') || '';
      if (category === 'all' || cardCategory.includes(category)) {
        card.style.display = 'flex';
      } else {
        card.style.display = 'none';
      }
    });
  }
